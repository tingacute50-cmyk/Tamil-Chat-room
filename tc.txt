    <script>
        const ageSelect = document.getElementById('reg-age');
        for (let age = 18; age <= 99; age++) {
            let opt = document.createElement('option');
            opt.value = age;
            opt.innerHTML = age;
            ageSelect.appendChild(opt);
        }

        let database = { username: "", password: "", gender: "", age: "" };
        let countdownValue = 30;
        let adminTimerInterval = null;
        let currentGeneratedCode = ""; // Holds the dynamic tracking token value

        // 7-Character Alphanumeric Generator Matrix
        function generateRandomTCode() {
            const chars = "ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789";
            let code = "";
            for (let i = 0; i < 7; i++) {
                code += chars.charAt(Math.floor(Math.random() * chars.length));
            }
            return code;
        }

        function switchScreen(screenId) {
            document.querySelectorAll('.screen').forEach(s => s.classList.remove('active'));
            document.getElementById(screenId).classList.add('active');
        }

        function handleRegister() {
            const u = document.getElementById('reg-username').value.trim();
            const p = document.getElementById('reg-password').value.trim();
            const g = document.getElementById('reg-gender').value;
            const a = document.getElementById('reg-age').value;
            
            if(u && p && g && a) {
                database.username = u;
                database.password = p;
                database.gender = g;
                database.age = a;
                alert(`Account created successfully!`);
                switchScreen('login-screen');
            } else {
                alert("Please fill out all fields.");
            }
        }

        function handleLogin() {
            const u = document.getElementById('login-username').value.trim();
            const p = document.getElementById('login-password').value.trim();
            const errorDiv = document.getElementById('login-error');

            if(u === database.username && p === database.password && u !== "") {
                errorDiv.style.display = 'none';
                
                // Random token is generated immediately when logging in
                currentGeneratedCode = generateRandomTCode();
                document.getElementById('system-generated-tcode').innerText = currentGeneratedCode;
                
                alert("வாங்க வாங்க! (Vanga Vanga!) \nWelcome to Tamil World Chatroom!");
                enterTamilWorld();
            } else {
                errorDiv.style.display = 'block';
            }
        }

        function enterTamilWorld() {
            switchScreen('main-chat-screen');
            document.getElementById('waiting-banner').style.display = 'block';
            document.getElementById('admin-challenge').style.display = 'none';
            document.getElementById('tcode-challenge-input').value = "";
            document.getElementById('challenge-error').style.display = 'none';

            setTimeout(() => {
                triggerAdminChallengePopup();
            }, 5000);
        }

        function triggerAdminChallengePopup() {
            document.getElementById('waiting-banner').style.display = 'none';
            document.getElementById('admin-challenge').style.display = 'block';
            
            countdownValue = 30;
            document.getElementById('pop-timer').innerText = countdownValue;

            adminTimerInterval = setInterval(() => {
                countdownValue--;
                document.getElementById('pop-timer').innerText = countdownValue;

                if (countdownValue <= 0) {
                    clearInterval(adminTimerInterval);
                    executeKickSequence();
                }
            }, 1000);
        }

        function submitChallengeCode() {
            // Irrespective of what the user inputs, always flash the error message
            const challengeError = document.getElementById('challenge-error');
            challengeError.style.display = 'block';
            document.getElementById('tcode-challenge-input').value = "";
        }

        function executeKickSequence() {
            alert("ADMIN: tcode illama vantha thooki potu mithipen !!!");
            switchScreen('login-screen');
            location.reload();
        }
